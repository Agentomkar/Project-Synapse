!
! ================================================================
! SYNAPSE — Dist-SW-A (Admin Building)
! Model: Cisco 2960-24TT (L2)
! Carries VLANs: 10 (Admin), 20 (Faculty), 99 (Mgmt)
! Also hosts the AI-Server on VLAN 99
! ================================================================
!
enable
configure terminal
!
hostname Dist-SW-A
no ip domain-lookup
service password-encryption
enable secret SynapseDist2026!
!
vtp mode transparent
!
vlan 10
 name ADMIN
vlan 20
 name FACULTY
vlan 99
 name MGMT
exit
!
! Uplinks to both Core switches
interface GigabitEthernet0/1
 description Uplink to Core-SW-1
 switchport mode trunk
 switchport trunk allowed vlan 10,20,99
 no shutdown
!
interface GigabitEthernet0/2
 description Uplink to Core-SW-2 (redundant)
 switchport mode trunk
 switchport trunk allowed vlan 10,20,99
 no shutdown
!
! Downlinks to Access switches
interface FastEthernet0/1
 description To Access-A1
 switchport mode trunk
 switchport trunk allowed vlan 10,99
 no shutdown
!
interface FastEthernet0/2
 description To Access-A2
 switchport mode trunk
 switchport trunk allowed vlan 20,99
 no shutdown
!
! Direct port for AI-Server (VLAN 99)
interface FastEthernet0/24
 description AI-Server (SYNAPSE Orchestrator)
 switchport mode access
 switchport access vlan 99
 spanning-tree portfast
 no shutdown
!
! --- Security: Shutdown unused ports ---
interface range FastEthernet0/3-23
 shutdown
!
! --- Management IP ---
interface Vlan99
 ip address 10.99.0.10 255.255.255.0
 no shutdown
!
ip default-gateway 10.99.0.1
!
! --- Enable Spanning Tree PortFast + BPDU Guard on access ports ---
spanning-tree mode rapid-pvst
spanning-tree portfast bpduguard default
!
end
write memory
